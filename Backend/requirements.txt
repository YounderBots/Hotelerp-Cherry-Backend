# Pinned to the stack the services are actually developed and run against.
#
# The previous pins (fastapi==0.85.1, pydantic==1.10.2, SQLAlchemy==1.4.43) did
# not match the running environment: the controllers are written for Pydantic v2
# and SQLAlchemy 2.x, so a clean `pip install -r requirements.txt` produced an
# install that failed at import. These versions are the ones verified working.

fastapi==0.115.6
starlette==0.41.3
pydantic==2.10.4
uvicorn==0.34.0
watchfiles==1.0.3

SQLAlchemy==2.0.36
PyMySQL==1.1.1

httpx==0.28.1
httpcore==1.0.9

# 3.4.0 fixes CVE-2024-33663 (algorithm confusion) and CVE-2024-33664 (JWT
# decompression DoS). Both are reachable from token verification.
python-jose[cryptography]==3.4.0
PyJWT==2.10.1

bcrypt==4.2.1

# 0.0.20 fixes CVE-2024-24762 (ReDoS) and CVE-2024-53981 (allocation DoS) in
# multipart parsing, which the gateway proxy exercises on file uploads.
python-multipart==0.0.20

itsdangerous==2.2.0

# Schema changes are applied with Alembic (Backend/migrations). The models also
# expose create_all via DB_AUTO_CREATE, but that only ever ADDS tables and
# columns -- it never alters or drops one -- so it is a dev convenience only and
# defaults to off in production.
alembic==1.14.0

# openpyxl is the engine passed explicitly to pd.ExcelWriter(engine="openpyxl")
# in HotelServices/resources/nightauditController.py. pandas does NOT depend on
# it, so leaving it out makes every Night Audit Excel export fail at runtime on
# a clean install even though it works on machines that happen to have it.
pandas==2.3.3
openpyxl==3.1.5

# Each service loads its OWN .env at startup, from the service directory next to
# main.py (configs/__init__.py). There is no shared runtime .env.
python-dotenv==1.0.1

# Country-aware phone validation. `resources/validation.py` normalises a phone
# number to E.164 and checks it against the real numbering plan for its country
# via libphonenumber, which is Google's maintained metadata for every plan
# (C-086). Pinned because libphonenumber ships new plan metadata in each release:
# floating this would silently change which numbers the product accepts, and the
# 120-case matrix in tests/test_phone_validation.py is pinned to this version.
# The browser half of the same rule is `libphonenumber-js` in
# Frontend/package.json, so both halves read the same plans.
phonenumbers==9.0.40
